Ruby

class Invoice
  # Sums the lines, in cents.
  def total(lines)
    lines.sum { |l| l.quantity * l.unit_cents }
  end
end
puts "Total: #{Invoice.new.total([])}"

/// Renders a document to PDF bytes.
pub fn render(html: &str) -> Result<Vec<u8>, Error> {
    let doc = Document::parse(html)?;
    Ok(doc.print())
}
